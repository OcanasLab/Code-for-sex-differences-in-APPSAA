# Function to count the number of detected cells
count_cells <- function(matrix) {
  return(ncol(matrix))
}

# Function to compute total RNA counts (sum of UMI counts)
compute_total_rna <- function(matrix) {
  return(sum(matrix))
}

# Compute percent retention and reduction

cell_rna_counts$Percent_RNA_Reduced     <- 100 - cell_rna_counts$Percent_RNA_Retained

  plot_dotplot_for_dataset <- function(seurat_obj, dataset_name, genes, pathway) {
    
    genes.present <- genes[genes %in% rownames(seurat_obj)]
    
    if (length(genes.present) == 0) {
      message(paste("No genes present for", pathway, "in", dataset_name))
      return(NULL)
    }
    
    p <- DotPlot(
      seurat_obj,
      features = genes.present,
      group.by = "celltype",
      cols = c("blue", "red"),
      dot.scale = 8   # controls max dot size
    ) +
      RotatedAxis() +
      ggtitle(paste(dataset_name, "-", pathway)) +
      
      # STANDARDIZE % EXPRESSION (dot size)
      scale_size(
        range = c(0, 8),                     # visual size range
        breaks = c(0, 25, 50, 75, 100),     # legend values
        limits = c(0, 100)
      ) +
      
      # STANDARDIZE EXPRESSION SCALE
      scale_color_gradientn(
        colours = c("blue", "white", "red"),
        limits = c(-2, 2),                  # FIXED SCALE
        breaks = c(-2, -1, 0, 1, 2),
        oob = scales::squish                # squish values outside range
      ) +
      
      labs(
        size = "% Expressed",
        color = "Scaled Expression"
      )
    
    ggsave(
      filename = paste0("DotPlot_", dataset_name, "_", pathway, ".png"),
      plot = p,
      width = 12,
      height = 6
    )
  }
