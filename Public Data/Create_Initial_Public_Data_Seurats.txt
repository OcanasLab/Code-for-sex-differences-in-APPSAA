# 1. Extract the raw, unnormalized counts matrix from your original object
# (Assuming your original object is named 'seurat_ensembl')
counts_matrix <- GetAssayData(GSE156183_Seurat, assay = "RNA", layer = "counts")

# 2. Clean Ensembl IDs (strips version decimals like ENSG00000139618.15 -> ENSG00000139618)
cleaned_ids <- gsub("\\..*", "", rownames(counts_matrix))


mart <- useMart(biomart = "ensembl", dataset = "mmusculus_gene_ensembl")

symb <- getBM(attributes = c("ensembl_gene_id","mgi_symbol"),
              filters = "ensembl_gene_id", cleaned_ids,
              mart = mart)
symbs <- symb$mgi_symbol[match(cleaned_ids, symb$ensembl_gene_id, nomatch = NA)]
num <- length(symbs)
# 4. Handle missing/NA values or duplicates to prevent Seurat errors
# Replace missing symbols with the original Ensembl ID

for (k in 1:num){ifelse(symbs[k] == "",symbs[k]<-NA, symbs)}
symbs <- ifelse(is.na(symbs), cleaned_ids, symbs)


# Make names unique if multiple Ensembl IDs mapped to the same Symbol
symbs <- make.unique(symbs)

#5. Assign the new gene symbols to the counts matrix rows
rownames(counts_matrix) <- symbs

# 6. Create a brand-new Seurat object with the updated matrix
GSE156183_Seurat1 <- CreateSeuratObject(counts = counts_matrix)

# 7. Optional: Copy over your original metadata if you had any
GSE156183_Seurat1@meta.data <- GSE156183_Seurat@meta.data


Idents(GSE156183_Seurat1) <- "sample"
